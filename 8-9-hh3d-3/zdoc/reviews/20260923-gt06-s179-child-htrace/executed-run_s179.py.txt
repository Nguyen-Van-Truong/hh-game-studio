def pump(stream, path, rows):
    with path.open("w", encoding="utf-8", errors="replace") as out:
        for line in iter(stream.readline, ""):
            rows.append(line.rstrip("\r\n")); out.write(line); out.flush()

def children(parent):
    cmd = ('$p=Get-CimInstance Win32_Process -Filter "ParentProcessId=%d" | '
           'Select-Object ProcessId,ParentProcessId,ExecutablePath; '
           '$p | ConvertTo-Json -Compress') % parent
    raw = subprocess.check_output(["powershell.exe","-NoProfile","-Command",cmd], text=True, timeout=5).strip()
    if not raw: return []
    value = json.loads(raw); return value if isinstance(value, list) else [value]

def find_child(parent):
    expected = os.path.normcase(str(CHILD_EXE))
    for row in children(parent):
        if row.get("ExecutablePath") and os.path.normcase(row["ExecutablePath"]) == expected:
            return {"pid": int(row["ProcessId"]), "parent_pid": int(row["ParentProcessId"]),
                    "executable": row["ExecutablePath"],
                    "sha256": hashlib.sha256(CHILD_EXE.read_bytes()).hexdigest()}
    return None

def owned(pid, executable):
    try:
        value = subprocess.check_output(["powershell.exe","-NoProfile","-Command",
            '(Get-CimInstance Win32_Process -Filter "ProcessId=%d").ExecutablePath' % pid],
            text=True, timeout=5).strip()
    except Exception: return False
    return os.path.normcase(value) == os.path.normcase(str(executable))
